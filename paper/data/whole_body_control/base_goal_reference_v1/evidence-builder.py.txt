"""Compact source-bound evidence; raw polynomials/traces/weights remain ignored."""
import hashlib,json,math,shutil
from pathlib import Path
import numpy as np
ROOT=Path.cwd();OUT=ROOT/'paper/data/whole_body_control/base_goal_reference_v1'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):return json.loads(p.read_text())
def write(p,d):p.write_text(json.dumps(d,indent=2,allow_nan=False)+'\n')
KEYS=('base_goal_error','base_heading_error_rad','ee_position_error','ee_orientation_error','time_to_first_settle','settled_seconds','base_linear_norm_m_s','base_angular_norm_rad_s','peak_hold_base_drift_m','peak_contact_force','max_penetration','unsafe_contact','wheel_lift','peak_base_linear_jerk_m_s3','peak_base_yaw_jerk_rad_s3')
def render(value,depth=0,key=None):
 indent='  '*depth
 if isinstance(value,dict):
  return '{\n'+',\n'.join('  '*(depth+1)+json.dumps(k)+': '+render(v,depth+1,k) for k,v in value.items())+'\n'+indent+'}'
 if isinstance(value,list):
  if key=='cases':
   return '[\n'+',\n'.join('  '*(depth+1)+json.dumps(row,separators=(',', ': '),allow_nan=False) for row in value)+'\n'+indent+']'
  if all(not isinstance(item,(dict,list)) for item in value):return json.dumps(value,allow_nan=False)
  return '[\n'+',\n'.join('  '*(depth+1)+render(row,depth+1) for row in value)+'\n'+indent+']'
 return json.dumps(value,allow_nan=False)

def compact(c):
 out={k:c[k] for k in ('id','seed','mode','cohort','success','code','target','start','duration_s','admission','endpoint_xy_error_m','endpoint_heading_error_rad','stop_confirmed','primitive','boundary_reached','terminal_pose_in_contract','gripper_open','full_pick_place_success','inference_counts','safety_flags','history_preserved','switched','source_case') if k in c}
 if 'case' in c:out['case']=c['case']
 m=c.get('metrics',c.get('physical',{}));out['metrics']={k:m[k] for k in KEYS if k in m}
 if c.get('trace') and 'target' in c:
  error=np.array(c['trace'][-1]['base'])-c['target'];error[2]=math.atan2(math.sin(error[2]),math.cos(error[2]));out['endpoint_xy_error_m']=float(np.linalg.norm(error[:2]));out['endpoint_heading_error_rad']=float(abs(error[2]));out['final_reference_progress_s']=c['trace'][-1].get('progress')
 return out

  runner=ROOT/f'logs/recovery/base-point-reference-probe-v{version}/run.py';shutil.copyfile(runner,OUT/f'probe-v{version}.py.txt')
 cohorts={}

 wb=native/'wandb-evaluation.json'
 if wb.exists():report['wandb_evaluation']=read(wb)

 print(json.dumps({'summary':run['summary'],'regressions':report['regressions'],'artifact_bytes':sum(p.stat().st_size for p in OUT.iterdir())}))
