A positive projection gap proves separation; this is not exact Euclidean clearance.
No live physics changes or additional rollouts. Retargeted nominal paths are
reported separately from the initial reference.
"""

def shape_vertices(m,d,g,frame):
 kind=int(m.geom_type[g]);R=d.geom_xmat[g].reshape(3,3).copy();t=d.geom_xpos[g].copy();F=d.geom_xmat[frame].reshape(3,3).copy();origin=d.geom_xpos[frame].copy()
 if kind==mujoco.mjtGeom.mjGEOM_BOX:
  local=np.array([[x,y,z] for x in (-1,1) for y in (-1,1) for z in (-1,1)])*m.geom_size[g]
  axes=(F.T@R).T
 elif kind==mujoco.mjtGeom.mjGEOM_MESH:
  index=int(m.geom_dataid[g]);a,n=int(m.mesh_vertadr[index]),int(m.mesh_vertnum[index]);local=m.mesh_vert[a:a+n].copy()
  f,k=int(m.mesh_faceadr[index]),int(m.mesh_facenum[index]);faces=m.mesh_face[f:f+k].copy();tri=local[faces]
  normals=np.cross(tri[:,1]-tri[:,0],tri[:,2]-tri[:,0]);normals=normals[np.linalg.norm(normals,axis=1)>1e-12];normals/=np.linalg.norm(normals,axis=1)[:,None];axes=normals@R.T@F
 else:raise ValueError('unexpected collision proxy type')
 return local@R.T@F+(t-origin)@F,axes
