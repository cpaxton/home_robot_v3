def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def save(p,value):
 tmp=p.with_suffix('.partial');tmp.write_text(json.dumps(value,indent=2,default=json_value,allow_nan=False)+'\n');tmp.replace(p)
def make_general(cases,path,horizon):
 env=CoupledTrackingEnv('innate_mars',cases,suite_path=path)
 env.unwrapped.config=replace(env.unwrapped.config,horizon=horizon)
 return env

def warm_critics(learner,rollout):
 cfg=learner.config;p=learner.policy
 _,targets=grouped_gae(rollout['rewards'],rollout['values'],rollout['next_values'],rollout['terminated'],rollout['episode_end'],rollout['ticks'],gamma=cfg.gamma,lam=cfg.gae_lambda)
 obs={k:torch.as_tensor(v,dtype=torch.float32) for k,v in rollout['observations'].items()};targets=torch.from_numpy(targets)
 for _ in range(cfg.epochs):
  for indices in torch.randperm(len(targets)).split(cfg.batch_size):
   loss=.5*((p.values({k:v[indices] for k,v in obs.items()})-targets[indices])**2).mean(0).sum()
   learner.optimizer.zero_grad();loss.backward()
   for critic in p.critics:torch.nn.utils.clip_grad_norm_(critic.parameters(),cfg.max_grad_norm)
   learner.optimizer.step()
 return {'critic_only':True,'value_loss':float(loss.detach())}

  finally:env.close()

 learner=MultiCriticPPO(policy,config)

  row={'case_id':info.get('case_id'),'family':info.get('training_family'),'code':info.get('code'),'success':bool(info.get('primitive_success',False)),'stop_confirmed':info.get('stop_confirmed'),'wheel_lift':bool(info.get('wheel_lift',False)),'unsafe_contact':bool(info.get('unsafe_contact',False)),'invalid_physics':bool(info.get('invalid_physics',False)),'decisions_so_far':decisions}

   run.log({'decisions':decisions,'control_ticks':ticks,'actor_lr':record['actor_lr'],'tracking_reward':float(np.mean(rollout['rewards'][:,0])),'stability_reward':float(np.mean(rollout['rewards'][:,1])),'contact_reward':float(np.mean(rollout['rewards'][:,2]))},step=decisions)
   meta.update(decisions=decisions,control_ticks=ticks,new_ppo_actor_decisions=max(0,decisions-protocol['critic_only_initial_decisions']),mixture=env.ledger(),general_mixture=general.ledger(),episodes=len(episodes));save(OUT/'run.json',meta)
   if decisions%2048==0:print('PPO',decisions,'/',protocol['decisions'],'ticks',ticks,'episodes',len(episodes),flush=True)

    export_actor(policy.actor,directory/'actor',native.config,native.fingerprint,steps=decisions,training={'scope':'coupled_direction_ppo_v1','teacher_actions_in_loss':False,'source_weights_sha256':initializer.manifest['weights_sha256']})
  final=WholeBodyBundle(OUT/f'checkpoint-{decisions}/actor');meta['phase']='final_evaluation';save(OUT/'run.json',meta)
 finally:
  env.close();run.finish()
 summary={}
 # Reuse the tested evaluator without collecting any new audit labels.

 try:
  for path,name in ((SUITE,'final-v1-general'),(COARSE_SUITE,'final-v2-general')):
   rows=evaluate(final.actor,load_suite(path)['robots']['innate_mars'],path,name);summary[name]={'attempts':len(rows),'successes':sum(x['success'] for x in rows)}
  audits=json.loads((PRIOR/'audit-cases.json').read_text());rows=evaluate(final.actor,audits,COARSE_SUITE,'final-base-audit');summary['final-base-audit']={'attempts':len(rows),'successes':sum(x['success'] for x in rows)}
 finally:evaluate.__globals__['OUT']=old_out
 rows=primitive_rollout(development['train_cases']+development['audit_cases'],policy=final.actor,primitive='grasp_pose',planning_scope='unloaded_prefix',wheel_feedback_current_velocity=True);save(OUT/'final-tamp.json',rows);summary['final-tamp']={'attempts':len(rows),'successes':sum(x['success'] for x in rows)}
 assert digest(source_state(ROOT))==meta['source_state_digest']
 meta.update(complete=True,phase='complete',summary=summary);save(OUT/'run.json',meta);print('SUMMARY',summary,flush=True)
if __name__=='__main__':main()
