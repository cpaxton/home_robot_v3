def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def save(p,value):
 temp=p.with_suffix('.partial');temp.write_text(json.dumps(value,indent=2,default=json_value,allow_nan=False)+'\n');temp.replace(p)
def sensor(obs):return {k:np.asarray(obs[k],dtype=np.float32).copy() for k in ('history','preview','motion')}
def predict(actor,obs):
 with torch.no_grad():return actor({k:torch.from_numpy(v)[None] for k,v in sensor(obs).items()})[0].numpy().copy()
def stop(e):
 e.control.stop()
 for _ in range(16):
  if whole_body_stopped(e.control) or e._physics_invalid:break
  e._tick(np.zeros(e.profile.action_size))
 return bool(whole_body_stopped(e.control) and not e._physics_invalid)
def new_base_cases():return runpy.run_path(str(REC/'admit.py'))['cases']()
def audit_cases():
 motions=[(d,y) for d in (.12,-.12) for y in (.18,-.18)]+[(d,y) for d in (.18,-.18) for y in (.30,-.30)]+[(0.,y) for y in (.18,-.18,.30,-.30)]+[(d,0.) for d in (.12,-.12,.18,-.18)]
 return [{'id':f'innate_mars:base-direction-audit-v1:{i}','seed':35830000+i,'split':'audit','mode':'base_only','travel_m':d,'turn_rad':y,'arm_delta':[0.]*5,'base_mask':[True]*3} for i,(d,y) in enumerate(motions)]
def evaluate_general(actor,cases,suite_path,label):
 env=CoupledTrackingEnv('innate_mars',cases,suite_path=suite_path);rows=[];e=env.unwrapped
 try:
  for case in cases:
   try:
    obs,init=env.reset(options={'case_id':case['id']});trace=[]
    for tick in range(500):
     action=predict(actor,obs);obs,_,done,truncated,info=env.step(action)
     trace.append({'tick':tick,'action':action.tolist(),'base':e.control.base_pose().tolist()})
     if done or truncated:break
    metrics=e._info();confirmed=stop(e)
    row={'id':case['id'],'case':case,'success':bool(info.get('is_success')),'code':info['code'],'control_ticks':tick+1,'stop_confirmed':confirmed,'metrics':metrics,'reference_sha256':init['reference_sha256'],'trace':trace}
   except (ValueError,RuntimeError) as exc:row={'id':case['id'],'success':False,'code':str(exc),'initialization_failed':True}
   rows.append(row);save(OUT/(label+'.json'),rows);print('EVAL',label,case['id'],row['success'],row['code'],flush=True)
 finally:env.close()
 return rows

def collect_general(actor,cases,suite_path,drive,group,data):
 env=CoupledTrackingEnv('innate_mars',cases,suite_path=suite_path);e=env.unwrapped;summary=[]
 try:
  for c in cases:
   obs,_=env.reset(options={'case_id':c['id']});n=0
   for tick in range(500):
    label=task_pose_teacher_action(e.control).astype(np.float32)
    elapsed=e.control.trajectory_elapsed()-e.control.task_trajectory.end
    phase='moving' if elapsed<0 else 'settling' if elapsed<2 else 'late_hold'
    data[group].append((c['id'],phase,sensor(obs),label.copy()));n+=1
    action=label if drive=='teacher' else predict(actor,obs)
    obs,_,done,truncated,info=env.step(action)
    if done or truncated:break
   stop(e);summary.append({'group':group,'case':c['id'],'drive':drive,'labels':n,'code':info['code'],'success':bool(info.get('is_success'))})
 finally:env.close()
 return summary

def collect_tamp(actor,cases,drive,data):
 env=TampPrimitiveEnv(cases,oracle=True,primitive='grasp_pose',planning_scope='unloaded_prefix',wheel_feedback_current_velocity=True);summary=[]
 try:
  for c in cases:
   obs,info=env.reset(options={'case_id':c['id']});n=0
   for _ in range(2000):
    if 'oracle_action' not in info:break
    phase='announced_motion' if np.linalg.norm(obs['motion'])>1e-6 else 'announced_hold'
    label=np.asarray(info['oracle_action'],dtype=np.float32).copy();data['tamp'].append((c['id'],phase,sensor(obs),label));n+=1
    action=label if drive=='teacher' else predict(actor,obs)
    obs,_,done,truncated,info=env.step(action)
    if done or truncated:break
   summary.append({'group':'tamp','case':c['id'],'drive':drive,'labels':n,'code':info['code'],'success':bool(info.get('primitive_success'))})
 finally:env.close()
 return summary

def fit(actor,data,protocol,run):
 rng=np.random.default_rng(protocol['seed']);tensors={};index={}
 for group,rows in data.items():
  tensors[group]={k:torch.from_numpy(np.array([x[2][k] for x in rows])) for k in ('history','preview','motion')}
  tensors[group]['labels']=torch.from_numpy(np.array([x[3] for x in rows]))
  index[group]={}
  for i,(case,phase,_,_) in enumerate(rows):index[group].setdefault(case,{}).setdefault(phase,[]).append(i)
 # A fresh optimizer and zero weight decay make zero-gradient inactive rows
 # exactly invariant. Verify every frozen row/tensor again before exporting.
 for p in actor.parameters():p.requires_grad_(False)
 actor.gain.weight.requires_grad_(True);actor.gain.bias.requires_grad_(True)
 count=2*actor.feedback_width
 hooks=[]
 for parameter in (actor.gain.weight,actor.gain.bias):
  mask=torch.zeros_like(parameter);mask[:count]=1
  hooks.append(parameter.register_hook(lambda grad,mask=mask:grad*mask))
 opt=torch.optim.Adam([actor.gain.weight,actor.gain.bias],lr=protocol['learning_rate'])
 try:
  for step in range(protocol['supervised_updates']):
   batches={k:[] for k in ('history','preview','motion','labels')}
   for group,n in (('base',16),('general',16),('tamp',32)):
    ids=[];cases=list(index[group])
    for _ in range(n):
     c=cases[int(rng.integers(len(cases)))];phases=list(index[group][c]);phase=phases[int(rng.integers(len(phases)))];choices=index[group][c][phase];ids.append(choices[int(rng.integers(len(choices)))])
    for k in batches:batches[k].append(tensors[group][k][ids])
   batch={k:torch.cat(v) for k,v in batches.items()}
   prediction=actor(batch);mask=actor.action_mask(batch)[:,:2]
   loss=(((prediction[:,:2]-batch['labels'][:,:2])*mask)**2).sum()/mask.sum().clamp_min(1)
   opt.zero_grad();loss.backward();torch.nn.utils.clip_grad_norm_([actor.gain.weight,actor.gain.bias],1.);opt.step()
   if (step+1)%64==0:
    record={'supervised_update':step+1,'base_action_mse':float(loss.detach())};run.log(record,step=step+1)
    with (OUT/'updates.jsonl').open('a') as f:f.write(json.dumps(record)+'\n')
    print('FIT',record,flush=True)
 finally:
  for hook in hooks:hook.remove()
 return {group:{c:{p:len(rows) for p,rows in phases.items()} for c,phases in cs.items()} for group,cs in index.items()}

 evaluate_general(actor,audits,COARSE_SUITE,'initial-base-audit')
 # Evaluation may consume RNG; restart the declared training stream.
 np.random.seed(protocol['seed']);torch.manual_seed(protocol['seed'])
 data={'base':[],'general':[],'tamp':[]};collected=[]

 assert all(row[0] in train_ids for rows in data.values() for row in rows)

 meta['phase']='fitting';save(OUT/'run.json',meta)
 import wandb
 run=wandb.init(project='emet-mars-control',name='base-direction-bootstrap-v1',mode='offline',dir=str(OUT),config={**protocol,'source_state_digest':digest(sources)})
 try:exposure=fit(actor,data,protocol,run)
 finally:run.finish()
 count=2*actor.feedback_width
 for k,v in actor.state_dict().items():
  if k in ('gain.weight','gain.bias'):torch.testing.assert_close(v[count:],before[k][count:],rtol=0,atol=0)
  else:torch.testing.assert_close(v,before[k],rtol=0,atol=0)
 meta['frozen_arm_grip_encoder_context_exact']=True;save(OUT/'replay-exposure.json',exposure)
 env=CoupledTrackingEnv('innate_mars')
 try:final=export_actor(actor,OUT/'final/actor',env.unwrapped.config,env.unwrapped.fingerprint,steps=0,training={'scope':'base_output_only_supervised_fit','supervised_updates':protocol['supervised_updates'],'source_weights_sha256':original.manifest['weights_sha256'],'frozen_arm_grip_encoder_context_exact':True,'new_ppo_steps':0})
 finally:env.close()
 meta['phase']='final_evaluation';meta['candidate_weights_sha256']=final.manifest['weights_sha256'];save(OUT/'run.json',meta)
