def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):return json.loads(p.read_text())
def render(value,depth=0,key=None):
 indent='  '*depth
 if isinstance(value,dict):return '{\n'+',\n'.join('  '*(depth+1)+json.dumps(k)+': '+render(v,depth+1,k) for k,v in value.items())+'\n'+indent+'}'
 if isinstance(value,list):
  if key=='cases':return '[\n'+',\n'.join('  '*(depth+1)+json.dumps(row,separators=(',',': '),allow_nan=False) for row in value)+'\n'+indent+']'
  if all(not isinstance(v,(dict,list)) for v in value):return json.dumps(value,allow_nan=False)
  return '[\n'+',\n'.join('  '*(depth+1)+render(v,depth+1) for v in value)+'\n'+indent+']'
 return json.dumps(value,allow_nan=False)

 if 'error' in c:row['error']=c['error']
 return row
