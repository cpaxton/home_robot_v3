def read(path):return json.loads(path.read_text())
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def clean(row):
 result={k:v for k,v in row.items() if k not in {'trace','reference'}}
 if 'reference' in row:result['reference_sha256']=digest(row['reference'])
 if 'physics_trace' in result:
  result['physics_trace']=dict(result['physics_trace']);p=Path(result['physics_trace']['path']);parts=p.parts;p=Path(*parts[parts.index('logs'):]);assert sha(ROOT/p)==result['physics_trace']['sha256'];result['physics_trace']['path']=str(p)
 return result
